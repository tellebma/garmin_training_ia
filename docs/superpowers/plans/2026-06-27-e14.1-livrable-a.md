# E14.1 Livrable A — Graphes synchronisés Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Remplacer le graphe multi-lignes unique du détail d'activité par des graphes séparés par métrique, avec un curseur de survol synchronisé entre tous les graphes et un marqueur corrélé sur la carte GPS.

**Architecture:** Un conteneur client `ActivityDetailViz` détient un état `hoveredIndex: number | null` (index de sample) partagé entre N graphes recharts et la carte MapLibre. Un seul composant graphe configurable `SyncedMetricChart` (line ou area) évite la duplication. La carte du livrable B est étendue avec des props optionnelles pour afficher un marqueur au sample survolé.

**Tech Stack:** Next.js 15 (RSC + client components), TypeScript strict, recharts (existant), maplibre-gl (existant), Vitest + Testing Library.

## Global Constraints

- TypeScript strict — pas de `any`, props `readonly`, `noUncheckedIndexedAccess` actif.
- Pas de template literals avec nombres bruts (ESLint `restrict-template-expressions`) — utiliser `String()`.
- Graphes/canvas non accessibles : `aria-hidden` + alternative `sr-only`. **Jamais `role="img"`** (Sonar S6819).
- Pas d'emoji dans l'UI.
- `hoveredIndex` est un **index de sample** (`ActivitySample.sample_index` position dans le tableau), pas un pixel.
- Couleurs via tokens recharts existants (`var(--chart-1)`…`var(--chart-5)`, `var(--border)`, `var(--card)`).
- Composants MapLibre client-only ; la carte garde son contrat `dynamic(ssr:false)` via le lazy wrapper.
- Quality gate SonarQube : couverture du nouveau code ≥ 90 %, 0 violation, duplication ≤ 3 %. En cas d'échec, lire la condition via `api/qualitygates/project_status?projectKey=garmin_training_ia&branch=main` avec le token `SONAR_TOKEN_TELLEBMA` (dans `~/.bashrc`).
- `ActivitySample` (de `@/lib/coach/activity-analysis`) : `sample_index, sample_time, elapsed_s, distance_m, elevation_m, heart_rate_bpm, power_w, cadence_rpm, speed_m_s, latitude, longitude` (tous `number|null` sauf `sample_index:number`, `sample_time:string|null`).

```typescript
// lib/activity/chart-data.ts
import type { ActivitySample } from '@/lib/coach/activity-analysis'

export type ChartXKey = 'elapsed_s' | 'distance_m'

export interface ChartPoint {
  index: number
  x: number
  elevation_m: number | null
  heart_rate_bpm: number | null
  power_w: number | null
  cadence_rpm: number | null
  pace_s_per_km: number | null
}

function pace(speed: number | null): number | null {
  if (typeof speed !== 'number' || speed <= 0) return null
  return Math.round((1000 / speed) * 100) / 100
}

export function toChartPoints(samples: ActivitySample[], xKey: ChartXKey): ChartPoint[] {
  return samples.map((s, index) => ({
    index,
    x: typeof s[xKey] === 'number' ? (s[xKey] as number) : 0,
    elevation_m: s.elevation_m,
    heart_rate_bpm: s.heart_rate_bpm,
    power_w: s.power_w,
    cadence_rpm: s.cadence_rpm,
    pace_s_per_km: pace(s.speed_m_s),
  }))
}

export function hasMetric(samples: ActivitySample[], key: keyof ActivitySample): boolean {
  return samples.some((s) => typeof s[key] === 'number')
}
```

```tsx
// app/(app)/_components/charts/synced-metric-chart.tsx
'use client'

import {
  Area, AreaChart, CartesianGrid, Line, LineChart, ReferenceLine,
  ResponsiveContainer, Tooltip, XAxis, YAxis,
} from 'recharts'
import type { ChartPoint } from '@/lib/activity/chart-data'

export type MetricKey =
  | 'elevation_m' | 'heart_rate_bpm' | 'power_w' | 'cadence_rpm' | 'pace_s_per_km'

export interface SyncedMetricChartProps {
  readonly points: ChartPoint[]
  readonly metric: MetricKey
  readonly label: string
  readonly color: string
  readonly variant: 'line' | 'area'
  readonly hoveredIndex: number | null
  readonly onHover: (index: number | null) => void
  readonly height?: number
}

export function SyncedMetricChart({
  points, metric, label, color, variant, hoveredIndex, onHover, height = 140,
}: SyncedMetricChartProps) {
  const hoveredX =
    hoveredIndex !== null && hoveredIndex >= 0 && hoveredIndex < points.length
      ? points[hoveredIndex]?.x
      : undefined

  const handleMove = (state: { activeTooltipIndex?: number }) => {
    onHover(typeof state?.activeTooltipIndex === 'number' ? state.activeTooltipIndex : null)
  }

  const cursor =
    hoveredX !== undefined ? <ReferenceLine x={hoveredX} stroke={color} strokeDasharray="3 3" /> : null

  return (
    <figure aria-hidden="true" className="m-0">
      <figcaption className="text-muted-foreground mb-1 text-xs">{label}</figcaption>
      <ResponsiveContainer width="100%" height={height}>
        {variant === 'area' ? (
          <AreaChart data={points} onMouseMove={handleMove} onMouseLeave={() => onHover(null)}
            margin={{ top: 4, right: 8, left: -16, bottom: 0 }}>
            <CartesianGrid strokeDasharray="3 3" stroke="var(--border)" />
            <XAxis dataKey="x" tick={{ fontSize: 10 }} interval="preserveStartEnd" />
            <YAxis tick={{ fontSize: 10 }} width={36} />
            <Tooltip contentStyle={{ background: 'var(--card)', border: '1px solid var(--border)', borderRadius: 6, fontSize: 12 }} />
            <Area type="monotone" dataKey={metric} stroke={color} fill={color} fillOpacity={0.2} strokeWidth={2} dot={false} isAnimationActive={false} connectNulls />
            {cursor}
          </AreaChart>
        ) : (
          <LineChart data={points} onMouseMove={handleMove} onMouseLeave={() => onHover(null)}
            margin={{ top: 4, right: 8, left: -16, bottom: 0 }}>
            <CartesianGrid strokeDasharray="3 3" stroke="var(--border)" />
            <XAxis dataKey="x" tick={{ fontSize: 10 }} interval="preserveStartEnd" />
            <YAxis tick={{ fontSize: 10 }} width={36} />
            <Tooltip contentStyle={{ background: 'var(--card)', border: '1px solid var(--border)', borderRadius: 6, fontSize: 12 }} />
            <Line type="monotone" dataKey={metric} stroke={color} strokeWidth={2} dot={false} isAnimationActive={false} connectNulls />
            {cursor}
          </LineChart>
        )}
      </ResponsiveContainer>
    </figure>
  )
}
```

```tsx
// app/(app)/_components/activity/activity-detail-viz.tsx
'use client'

import { useMemo, useState } from 'react'
import { toChartPoints, hasMetric, type ChartXKey, type ChartPoint } from '@/lib/activity/chart-data'
import { SyncedMetricChart, type MetricKey } from '../charts/synced-metric-chart'
import { ActivityRouteMapLazy } from '../maps/activity-route-map-lazy'
import type { ActivitySample } from '@/lib/coach/activity-analysis'

interface MetricConfig {
  readonly metric: MetricKey
  readonly sampleKey: keyof ActivitySample
  readonly label: string
  readonly color: string
  readonly variant: 'line' | 'area'
}

const METRICS: readonly MetricConfig[] = [
  { metric: 'elevation_m', sampleKey: 'elevation_m', label: 'Altitude', color: 'var(--chart-2)', variant: 'area' },
  { metric: 'heart_rate_bpm', sampleKey: 'heart_rate_bpm', label: 'Fréquence cardiaque', color: 'var(--chart-1)', variant: 'line' },
  { metric: 'pace_s_per_km', sampleKey: 'speed_m_s', label: 'Allure', color: 'var(--chart-3)', variant: 'line' },
  { metric: 'power_w', sampleKey: 'power_w', label: 'Puissance', color: 'var(--chart-4)', variant: 'line' },
  { metric: 'cadence_rpm', sampleKey: 'cadence_rpm', label: 'Cadence', color: 'var(--chart-5)', variant: 'line' },
]

export function ActivityDetailViz({ samples }: { readonly samples: ActivitySample[] }) {
  const [hoveredIndex, setHoveredIndex] = useState<number | null>(null)
  const xKey: ChartXKey = hasMetric(samples, 'distance_m') ? 'distance_m' : 'elapsed_s'
  const points: ChartPoint[] = useMemo(() => toChartPoints(samples, xKey), [samples, xKey])
  const active = METRICS.filter((m) => hasMetric(samples, m.sampleKey))
  const hasGps = samples.some((s) => typeof s.latitude === 'number' && typeof s.longitude === 'number')

  return (
    <div className="space-y-3">
      <span className="sr-only">Graphiques de l&apos;activité (altitude, fréquence cardiaque, allure, puissance, cadence) synchronisés avec la carte.</span>
      {hasGps && <ActivityRouteMapLazy samples={samples} hoveredIndex={hoveredIndex} />}
      {active.map((m) => (
        <SyncedMetricChart key={m.metric} points={points} metric={m.metric} label={m.label}
          color={m.color} variant={m.variant} hoveredIndex={hoveredIndex} onHover={setHoveredIndex} />
      ))}
    </div>
  )
}
```
